<script>
  (function () {
    var search = document.getElementById('talks-search');
    var buttons = document.querySelectorAll('.talks-filter .tag');
    var items = document.querySelectorAll('.talks-list ol.bibliography > li');
    var headings = document.querySelectorAll('.talks-list h2.bibliography');
    var empty = document.getElementById('talks-empty');
    var activeFilter = 'all';

    function update() {
      var query = search.value.trim().toLowerCase();
      var shown = 0;
      items.forEach(function (item) {
        var matchesFilter = activeFilter === 'all' || item.querySelector('.tag-' + activeFilter);
        var matchesSearch = query === '' || item.textContent.toLowerCase().indexOf(query) !== -1;
        var visible = matchesFilter && matchesSearch;
        item.style.display = visible ? '' : 'none';
        if (visible) shown++;
      });
      headings.forEach(function (h) {
        var list = h.nextElementSibling;
        if (!list) return;
        var any = Array.prototype.some.call(list.children, function (li) {
          return li.style.display !== 'none';
        });
        h.style.display = any ? '' : 'none';
        list.style.display = any ? '' : 'none';
      });
      empty.hidden = shown !== 0;
    }

    buttons.forEach(function (button) {
      button.addEventListener('click', function () {
        activeFilter = button.getAttribute('data-filter');
        buttons.forEach(function (b) { b.classList.toggle('is-active', b === button); });
        update();
      });
    });

    search.addEventListener('input', update);
  })();
</script>
